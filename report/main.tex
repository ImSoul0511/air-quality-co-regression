\section{Cấu hình chung của hệ thống: \texttt{config.py}}

\texttt{config.py} là file cấu hình chung của toàn hệ thống. Nó định nghĩa các hằng số và hàm tiện ích được sử dụng xuyên suốt toàn bộ dự án.

\begin{itemize}
  \item \texttt{RANDOM\_STATE = 42}: Seed toàn cục đảm bảo tính tái lập của toàn bộ dự án từ sinh dữ liệu giả lập trong bởi các hàm hỗ trợ trong \texttt{utils.py}, Fisher-Yates shuffle trong \texttt{cross\_validation.py}\ldots
  \item \texttt{EPSILON = 1e-12}: Ngưỡng epsilon toàn cục. Mọi số có giá trị tuyệt đối nhỏ hơn ngưỡng này đều được coi là bằng 0 về mặt số học. Ngưỡng này được dùng nhất quán trong tất cả phép chia, kiểm tra pivot, và tính VIF để tránh chia cho 0.
  \item \texttt{is\_zero(x)}: Kiểm tra một số có bằng 0 không theo EPSILON.
  \item \texttt{zero\_rectify(x)}: Trả về 0.0 nếu giá trị của x đủ nhỏ. Tránh các giá trị gây nhiễu trong quá trình tính toán chấm động.
  \item \texttt{calculate\_relative\_error(A, x\_hat, b)}: Tính sai số tương đối (relative error) của nghiệm gần đúng $\hat{x}$ đối với hệ phương trình tuyến tính $A x = b$. Công thức:
    \[
    \frac{\|A \hat{x} - b\|}{\|b\|}
    \]
    Hàm được sử dụng để kiểm định chất lượng của OLS và Ridge trong test\_case.
\end{itemize}
Các hằng số và các hàm xử lý sai số có ý nghĩa quan trọng đảm bảo tính nhất quán của toàn hệ thống. Đặc biệt với \texttt{RANDOM\_STATE} có nhiệm vụ quan trọng trong việc quản lý trạng thái của việc sinh dữ liệu.

\section{Bài Toán Data Fitting}

\subsection{Phát biểu bài toán tổng quát}

\textbf{Định nghĩa 1.1} (Bài toán Data Fitting).
Cho tập dữ liệu
$\mathcal{D} = \{(\mathbf{x}_i, y_i)\}_{i=1}^{n}$
với $\mathbf{x}_i \in \mathbb{R}^p$, $y_i \in \mathbb{R}$.

Bài toán data fitting là tìm hàm
$f: \mathbb{R}^p \to \mathbb{R}$
trong một lớp hàm cho trước sao cho $f$
xấp xỉ tốt nhất ánh xạ từ $\mathbf{x}_i$ đến $y_i$
theo một tiêu chí đã định.

Trong mô hình \textbf{hồi quy tuyến tính}, ta giả thiết:

\begin{equation}
  y_i = \beta_0 + \beta_1 x_{i,1} + \beta_2 x_{i,2} + \cdots + \beta_p x_{i,p} + \epsilon_i
      = \mathbf{x}_i^T \boldsymbol{\beta} + \epsilon_i
\end{equation}

với $\boldsymbol{\beta} = (\beta_0, \beta_1, \dots, \beta_p)^T \in \mathbb{R}^{p+1}$
là vector tham số cần ước lượng và $\epsilon_i$ là nhiễu ngẫu nhiên.
Viết dưới dạng ma trận với $\mathbf{X} \in \mathbb{R}^{n \times (p+1)}$
(ma trận design có cột đầu toàn 1):

\begin{equation}
  \mathbf{y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\epsilon}
\end{equation}

Vector biến phụ thuộc mục tiêu ký hiệu là $\mathbf{y} \in \mathbb{R}^n$, vector sai số ngẫu nhiên là $\boldsymbol{\epsilon} \in \mathbb{R}^n$, và ma trận thiết kế $\mathbf{X} \in \mathbb{R}^{n \times (p+1)}$ sở hữu cấu trúc cột đầu tiên chứa toàn bộ giá trị số 1 nhằm phục vụ việc tính toán hệ số chặn intercept $\beta_0$:
\begin{equation}
    \mathbf{X} = \begin{bmatrix} 
    1 & x_{1,1} & x_{1,2} & \dots & x_{1,p} \\
    1 & x_{2,1} & x_{2,2} & \dots & x_{2,p} \\
    \vdots & \vdots & \vdots & \ddots & \vdots \\
    1 & x_{n,1} & x_{n,2} & \dots & x_{n,p}
    \end{bmatrix}
\end{equation}

\subsection{Các Giả Thiết Gauss-Markov}

\begin{tcolorbox}[

  title={Các giả thiết Gauss--Markov (GM1 -- GM5)},
  fonttitle=\bfseries,

\textbf{GM1. \ Tuyến tính:} $\mathbf{y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$.

\medskip
\textbf{GM2. \ Không hoàn hảo đa cộng tuyến:}
$\operatorname{rank}(\mathbf{X}) = p + 1$ (các cột độc lập tuyến tính).

\medskip
\textbf{GM3. \ Ngoại sinh:}
$\mathbb{E}[\boldsymbol{\varepsilon}\mid\mathbf{X}] = \mathbf{0}$,
tức $\mathbb{E}[\varepsilon_i\mid\mathbf{x}_i] = 0$.

\medskip
\textbf{GM4. \ Đồng phương sai:}
$\operatorname{Var}(\boldsymbol{\varepsilon}\mid\mathbf{X}) = \sigma^2\mathbf{I}_n$,
tức $\operatorname{Var}(\varepsilon_i) = \sigma^2$ và
$\operatorname{Cov}(\varepsilon_i,\varepsilon_j) = 0$ với $i \ne j$.

\medskip
\textbf{GM5. \ Phần dư Chuẩn:}
$\boldsymbol{\varepsilon}\mid\mathbf{X}\sim\mathcal{N}(\mathbf{0},\sigma^2\mathbf{I}_n)$.
\end{tcolorbox}

\section{Phương Pháp Ordinary Least Squares (OLS)}

\subsection{Hàm mất mát và nghiệm OLS}

Phương pháp Bình phương Tối thiểu Thông thường (OLS) thực hiện tìm kiếm vector tham số tối ưu $\hat{\boldsymbol{\beta}}$ thông qua mục tiêu tối thiểu hóa tổng bình phương các phần dư sai số (Residual Sum of Squares - RSS):
\begin{equation}
    RSS(\boldsymbol{\beta}) = \|\mathbf{y} - \mathbf{X}\boldsymbol{\beta}\|_2^2 = \mathbf{y}^T\mathbf{y} - 2\boldsymbol{\beta}^T\mathbf{X}^T\mathbf{y} + \boldsymbol{\beta}^T\mathbf{X}^T\mathbf{X}\boldsymbol{\beta}
\end{equation}

Thực hiện lấy đạo hàm riêng của hàm mất mát $RSS$ theo vector tham số $\boldsymbol{\beta}$ và đặt giá trị phương trình đạo hàm thu được bằng giá trị $\mathbf{0}$ nhằm tìm điểm cực tiểu:
\begin{equation}
    \nabla_{\boldsymbol{\beta}}RSS = -2\mathbf{X}^T\mathbf{y} + 2\mathbf{X}^T\mathbf{X}\boldsymbol{\beta} = \mathbf{0}
\end{equation}
\begin{equation}
    \implies \mathbf{X}^T\mathbf{X}\boldsymbol{\beta} = \mathbf{X}^T\mathbf{y}
\end{equation}

Hệ phương trình trên được gọi là hệ phương trình chuẩn (Normal Equations). Do thỏa mãn giả thiết ma trận đặc trưng $\mathbf{X}$ đạt hạng đầy đủ cột, ma trận vuông tích số $\mathbf{X}^T\mathbf{X}$ đối xứng xác định dương và chắc chắn khả nghịch. Vì vậy nghiệm OLS duy nhất được biểu diễn bằng dạng đóng:
\begin{equation}
    \hat{\boldsymbol{\beta}}_{OLS} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{y}
\end{equation}

\textbf{Cài đặt Python}

Hàm \texttt{ols\_fit()} trong \texttt{part1/ols\_implementation.py} cài đặt nghiệm OLS từ hệ phương trình chuẩn. Ma trận đặc trưng ban đầu chưa có cột bias; hàm sẽ tự thêm cột intercept, tính $\mathbf{X}^T\mathbf{X}$, $\mathbf{X}^T\mathbf{y}$, sau đó suy ra hệ số hồi quy, fitted values, phần dư và ước lượng phương sai nhiễu.

\begin{lstlisting}[language=Python]
def ols_fit(X: list[list[float]], y: list[float]) -> dict:
    n = len(y)
    p = len(X[0])
    X_bias = add_bias_column(X)

    X_bias_T = transpose(X_bias) # Ma tran X_bias^T
    A = matmul(X_bias_T, X_bias) # Ma tran X_bias^T * X_bias
    b = matvec(X_bias_T, y) # Vector X_bias^T * y

    A_inv = inverse(A) # Ma tran A nghich đao
    beta_hat = matvec(A_inv, b) # Uoc luong he so beta

    y_hat = matvec(X_bias, beta_hat) # Gia tri du đoán
    residuals = vector_sub(y, y_hat) # Phan du: y - y_hat
    RSS = 0.0
    for val in residuals:
        RSS += val * val

    df = n - p - 1
    if df <= 0:
        raise ValueError("So mau phai lon hon so he so")
    sigma2_hat = RSS / df
    return {
        "beta_hat":   beta_hat,     # Vector trong so
        "sigma2_hat": sigma2_hat,   # Phuong sai nhieu
        "y_hat":      y_hat,        # Fitted values
        "residuals":  residuals,    # Phan du: y - y_hat
    }
\end{lstlisting}
\subsection{Ma Trận Chiếu và Hat Matrix}

Ma trận Hat (ký hiệu là $\mathbf{H}$) đại diện cho một ma trận chiếu vuông kích thước $n \times n$, đảm nhận vai trò hình học biến đổi trực tiếp từ vector dữ liệu thực tế $\mathbf{y}$ sang vector kết quả dự báo $\hat{\mathbf{y}}$:
\begin{equation}
    \hat{\mathbf{y}} = \mathbf{X}\hat{\boldsymbol{\beta}}_{OLS} = \mathbf{X}(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{y} = \mathbf{H}\mathbf{y}
\end{equation}
\begin{equation}
    \implies \mathbf{H} = \mathbf{X}(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T \in \mathbb{R}^{n \times n}
\end{equation}

Ma trận Hat sở hữu các đặc tính đại số tuyến tính cốt lõi sau đây:
\begin{itemize}
    \item \textbf{Tính đối xứng:} $\mathbf{H}^T = \mathbf{H}$.
    \item \textbf{Tính lũy đẳng (Idempotent):} $\mathbf{H}^2 = \mathbf{H}$.
    \item \textbf{Giá trị riêng:} Toàn bộ các giá trị riêng của ma trận lũy đẳng chỉ nhận giá trị là 0 hoặc 1.
    \item \textbf{Hạng ma trận:} Hạng của ma trận chiếu bằng vết trace: $\text{rank}(\mathbf{H}) = p+1$.
    \item \textbf{Vector phần dư:} Xác định qua ma trận bù đối góc: $\hat{\boldsymbol{\epsilon}} = (\mathbf{I}_n - \mathbf{H})\mathbf{y} $.
\end{itemize}

\subsection{Định Lý Gauss-Markov}

Định lý Gauss-Markov chứng minh toán học rằng: Khi đáp ứng trọn vẹn các giả thiết từ \textbf{GM1} đến \textbf{GM4}, ước lượng bình phương tối thiểu $\hat{\boldsymbol{\beta}}_{OLS}$ chính là ước lượng tuyến tính không chệch tốt nhất (BLUE - Best Linear Unbiased Estimator). Ý nghĩa cụ thể gồm:

\begin{itemize}
    \item \textbf{Tính không chệch:} Kỳ vọng toán học của ước lượng bằng chính giá trị tham số gốc: $\mathbb{E}[\hat{\boldsymbol{\beta}}_{OLS}] = \boldsymbol{\beta}$.
    \item \textbf{Tính hiệu quả tối ưu:} Trong tập hợp mọi ước lượng tuyến tính không chệch khác, ước lượng OLS đạt mức phương sai bé nhất: $\text{Var}(\tilde{\beta}_j) \ge \text{Var}(\hat{\beta}_j^{OLS})$.
\end{itemize}

Ma trận hiệp phương sai chính xác của ước lượng OLS được xác định qua công thức:
\begin{equation}
    \text{Var}(\hat{\boldsymbol{\beta}}_{OLS}|\mathbf{X}) = \sigma^2(\mathbf{X}^T\mathbf{X})^{-1}
\end{equation}
\textbf{Minh họa Monte Carlo - Định lý Gauss-Markov.}
Module \texttt{gauss\_markov\_demo.py} hiện thực một thí nghiệm Monte Carlo nhằm quan sát trực quan nội dung của định lý Gauss-Markov. Hàm chính \texttt{run\_gauss\_markov\_simulation()} nhận vào số lần mô phỏng \texttt{n\_sim}, số quan sát \texttt{n\_obs}, vector hệ số thật \texttt{true\_beta} và độ lệch chuẩn nhiễu \texttt{true\_sigma}. Nếu không truyền hệ số thật, chương trình sử dụng mặc định:
\[
    \boldsymbol{\beta} = (2.0,\,-1.5,\,0.8).
\]

Quy trình mô phỏng được tổ chức như sau:
\begin{enumerate}
    \item Cố định seed ngẫu nhiên bằng \texttt{random.seed(RANDOM\_STATE)} để bảo đảm tính tái lập.
    \item Sinh một ma trận đặc trưng cố định $\mathbf{X}$ với các phần tử lấy từ phân phối chuẩn chuẩn hóa. Sau đó thêm cột bias để tạo ma trận thiết kế.
    \item Tính trước phần tín hiệu không nhiễu:
    \[
        \mathbf{y}_{pure} = \mathbf{X}\boldsymbol{\beta}.
    \]
    \item Xây dựng một ước lượng thay thế có dạng Ridge với hệ số phạt $0.5$:
    \[
        \hat{\boldsymbol{\beta}}_{alt}
        =
        \left(\mathbf{X}^T\mathbf{X} + 0.5\mathbf{I}\right)^{-1}
        \mathbf{X}^T\mathbf{y}.
    \]
    Đây là một estimator tuyến tính nhưng có chệch do thêm thành phần phạt Ridge.
    \item Với mỗi lần mô phỏng, sinh nhiễu Gaussian $\boldsymbol{\epsilon}$, tạo dữ liệu mục tiêu
    \[
        \mathbf{y}_{sim} = \mathbf{y}_{pure} + \boldsymbol{\epsilon},
    \]
    rồi ước lượng hệ số bằng hai cách: OLS thông qua \texttt{ols\_fit()} và estimator thay thế thông qua ma trận trọng số đã tính trước.
    \item Lưu lại toàn bộ các vector hệ số ước lượng vào \texttt{beta\_ols\_list} và \texttt{beta\_alt\_list}, sau đó trả về cùng với vector hệ số thật.
\end{enumerate}

Ý nghĩa của thí nghiệm là so sánh phân phối thực nghiệm của hệ số OLS với hệ số từ estimator thay thế sau nhiều lần lặp. Khi dữ liệu thỏa các giả thiết Gauss-Markov, OLS dao động quanh hệ số thật và thể hiện tính không chệch. Estimator kiểu Ridge có thể làm giảm độ dao động trong một số trường hợp, nhưng đổi lại bị chệch vì hệ số bị kéo về gần 0. Điều này minh họa đúng tinh thần của định lý Gauss-Markov: OLS là ước lượng tuyến tính không chệch tốt nhất trong lớp các estimator tuyến tính không chệch, chứ không khẳng định mọi estimator có chệch đều kém hơn về mọi tiêu chí.

\subsection{Ước Lượng Phương Sai Nhiễu}

Do phương sai sai số $\sigma^2$ không thể biết trước trong thực tế, một ước lượng không chệch hiệu quả của đại lượng này được tính toán dựa trên tổng bình phương phần dư $RSS$ chia cho số bậc tự do hiệu chỉnh hệ thống:
\begin{equation}
    \hat{\sigma}^2 = \frac{RSS}{n - p - 1} = \frac{\|\mathbf{y} - \mathbf{X}\hat{\boldsymbol{\beta}}\|^2}{n - p - 1}
\end{equation}

\section{Đánh Giá Mô Hình Hồi Quy}

\subsection{Hệ số xác định $R^2$ và $R^2$ hiệu chỉnh}

Hệ số xác định $R^2$ biểu thị tỷ lệ phần trăm sự biến động của biến phụ thuộc mục tiêu được giải thích một cách thành công nhờ các biến độc lập trong mô hình:
\begin{equation}
    R^2 = 1 - \frac{RSS}{TSS} = 1 - \frac{\sum_{i}(y_i - \hat{y}_i)^2}{\sum_{i}(y_i - \bar{y})^2}
\end{equation}

Vì chỉ số $R^2$ luôn tăng lên một cách cơ học khi ta bổ sung thêm biến số mới, ta sử dụng hệ số xác định hiệu chỉnh $\bar{R}^2$ để phạt mô hình tương ứng với số lượng biến đặc trưng gia tăng:
\begin{equation}
    \bar{R}^2 = 1 - \frac{n - 1}{n - p - 1}(1 - R^2)
\end{equation}

\subsection{Kiểm Định Giả Thuyết Thống Kê}

Dựa trên giả thiết phân phối chuẩn \textbf{GM5}, các phép kiểm định ý nghĩa thống kê được triển khai cụ thể:

\begin{itemize}
    \item \textbf{Kiểm định t:} Đánh giá ý nghĩa riêng lẻ từng biến độc lập với giả thuyết $H_0: \beta_j = 0$:
    \begin{equation}
        t_j = \frac{\beta_j}{\hat{\sigma}\sqrt{[(X^TX)^{-1}]_{jj}}} \sim t_{n-p-1}
    \end{equation}
    \item \textbf{Kiểm định F:} Đánh giá mức độ ý nghĩa tổng thể toàn bộ mô hình hồi quy với giả thuyết $H_0: \beta_1 = \dots = \beta_p = 0$:
    \begin{equation}
        F = \frac{(TSS - RSS) / p}{RSS / (n - p - 1)} \sim F_{p, n-p-1}
    \end{equation}
\end{itemize}
\textbf{Nhiệm vụ của hàm \texttt{coef\_inference}.}
Sau khi đã huấn luyện mô hình OLS và thu được vector hệ số $\hat{\boldsymbol{\beta}}$, hàm \texttt{coef\_inference()} có nhiệm vụ xây dựng bảng suy diễn thống kê cho từng hệ số hồi quy. Cụ thể, hàm nhận vào ma trận đặc trưng \texttt{X}, vector mục tiêu \texttt{y}, hệ số ước lượng \texttt{beta\_hat} và phương sai nhiễu ước lượng \texttt{sigma2}. Từ đó, hàm tính ma trận hiệp phương sai của hệ số:
\[
    \operatorname{Cov}(\hat{\boldsymbol{\beta}})
    =
    \hat{\sigma}^2(\mathbf{X}^T\mathbf{X})^{-1},
\]
suy ra sai số chuẩn, thống kê $t$, p-value hai phía và khoảng tin cậy 95\% cho từng hệ số. Phiên bản hiện tại không dùng \texttt{scipy} hay \texttt{pandas}; p-value và giá trị tới hạn được tính thông qua các hàm phân phối Student-t tự cài đặt \texttt{\_t\_cdf()} và \texttt{\_t\_ppf()}. Kết quả cuối cùng được trả về dưới dạng \texttt{dict} thuần Python, gồm các khóa \texttt{index}, \texttt{coef}, \texttt{std\_err}, \texttt{t\_stat}, \texttt{p\_value}, \texttt{ci\_lower} và \texttt{ci\_upper}.
\pagebreak
\begin{lstlisting}[language=Python, caption={Tinh suy dien thong ke cho he so hoi quy}]
def coef_inference(X, y, beta_hat, sigma2):
    n = len(X)
    p = len(X[0])

    # Them cot bias neu X chua co cot intercept
    if len(X[0]) == len(beta_hat):
        X_bias = X
        p = len(X[0]) - 1
    elif len(X[0]) == len(beta_hat) - 1:
        X_bias = add_bias_column(X)
        p = len(X[0])
    else:
        raise ValueError(
            f"Bat thuong: So cot cua X ({len(X[0])}) khong khop "
            f"voi beta_hat ({len(beta_hat)})"
        )

    # Cov(beta_hat) = sigma2 * (X^T X)^{-1}
    XT = transpose(X_bias)
    XTX = matmul(XT, X_bias)
    XTX_inv = inverse(XTX)

    # Tinh sai so chuan SE_i
    std_errs = [math.sqrt(sigma2 * XTX_inv[i][i]) for i in range(len(XTX_inv))]

    # Tinh thong ke t_i = beta_i / SE_i
    t_stats = [b / se if se != 0.0 else 0.0 for b, se in zip(beta_hat, std_errs)]

    dof = n - p - 1

    # p-value hai phia dung Student-t tu cai dat
    p_values = [2.0 * (1.0 - _t_cdf(abs(t), dof)) for t in t_stats]

    # Gia tri toi han cho khoang tin cay 95%
    t_crit = _t_ppf(0.975, dof)

    ci_lower = [b - t_crit * se for b, se in zip(beta_hat, std_errs)]
    ci_upper = [b + t_crit * se for b, se in zip(beta_hat, std_errs)]

    index = ['intercept'] + [f'x{i+1}' for i in range(len(beta_hat) - 1)]
    return {
        'index':    index,
        'coef':     list(beta_hat),
        'std_err':  std_errs,
        't_stat':   t_stats,
        'p_value':  p_values,
        'ci_lower': ci_lower,
        'ci_upper': ci_upper,
    }
\end{lstlisting}
\section{Các Kỹ Thuật Nâng Cao trong Data Fitting}

\subsection{Đa cộng tuyến và Hệ số VIF}

Hiện tượng đa cộng tuyến xuất hiện khi tồn tại mối tương quan tuyến tính mạnh giữa các cột đặc trưng trong ma trận thiết kế, khiến ma trận tích rơi vào trạng thái gần suy biến. Để định lượng và phát hiện, ta áp dụng hệ số phóng đại phương sai (Variance Inflation Factor):
\begin{equation}
    VIF_j = \frac{1}{1 - R_j^2}
\end{equation}
Trong đó chỉ số $R_j^2$ là hệ số xác định có được từ việc hồi quy đặc trưng thứ $j$ theo toàn bộ các biến độc lập còn lại hệ thống. Ngưỡng giá trị $VIF > 10$ thể hiện hiện tượng đa cộng tuyến nghiêm trọng.

\subsection{Hồi Quy Regularization: Ridge và Lasso}

Kỹ thuật Regularization thực hiện bổ sung thành phần hình phạt điều chỉnh vào hàm mục tiêu tối ưu nhằm giải quyết bài toán đa cộng tuyến và quá nhiều chiều đặc trưng:

\begin{itemize}
    \item \textbf{Hồi quy Ridge ($L_2$):} Thu nhỏ giá trị các hệ số hồi quy tiến dần về mức 0 bằng cách phạt bình phương chuẩn Euclid:
    \begin{equation}
        \hat{\boldsymbol{\beta}}_{ridge} = \arg\min_{\boldsymbol{\beta}} \left\{ \|\mathbf{y} - \mathbf{X}\boldsymbol{\beta}\|_2^2 + \lambda \|\boldsymbol{\beta}\|_2^2 \right\} = (\mathbf{X}^T\mathbf{X} + \lambda \mathbf{I})^{-1}\mathbf{X}^T\mathbf{y}
    \end{equation}
    \item \textbf{Hồi quy Lasso ($L_1$):} Phạt dựa trên tổng giá trị tuyệt đối, có khả năng triệt tiêu hoàn toàn một số hệ số về đúng bằng 0 giúp tự động lựa chọn đặc trưng:
    \begin{equation}
        \hat{\boldsymbol{\beta}}_{lasso} = \arg\min_{\boldsymbol{\beta}} \left\{ \|\mathbf{y} - \mathbf{X}\boldsymbol{\beta}\|_2^2 + \lambda \|\boldsymbol{\beta}\|_1 \right\}
    \end{equation}
\end{itemize}

\subsection{Phân Trích Phần Dư (Residual Analysis)}

Phân tích phần dư phục vụ mục tiêu kiểm tra tính hợp lệ của hệ thống giả thiết Gauss-Markov qua 4 biểu đồ trực quan hình học:
\begin{enumerate}
    \item \textbf{Residuals vs Fitted:} Đánh giá tính chất quan hệ tuyến tính cùng tính đồng phương sai.
    \item \textbf{Normal Q-Q Plot:} Đo lường mức độ tuân thủ quy luật phân phối chuẩn của phần dư.
    \item \textbf{Scale-Location Plot:} Kiểm định tính đồng đều phương sai sai số trên toàn dải dự báo.
    \item \textbf{Cook's Distance:} Xác định các điểm quan sát dị biệt mang sức ảnh hưởng lớn.
\end{enumerate}

\subsection{Cross-Validation}

Phương pháp kiểm chứng chéo $k$-Fold thực hiện phân chia ngẫu nhiên tập dữ liệu thành $k$ phần bằng nhau. Quá trình lặp kiểm tra diễn ra $k$ lần liên tiếp, tính toán điểm số sai số bình phương trung bình trung vị hệ thống:
\begin{equation}
    CV_{(k)} = \frac{1}{k}\sum_{i=1}^k MSE_i
\end{equation}

\subsection{Hàm hồi quy tuyến tính gốc: \texttt{ols\_fit}}
\begin{lstlisting}[language=Python]
def ols_fit(X, y):
    X_T = transpose(X)
    XtX = matmul(X_T, X)
    inv_XtX = inverse(XtX)
    Xty = matvec(X_T, y)
    beta_hat = matvec(inv_XtX, Xty)
    
    y_hat = matvec(X, beta_hat)
    residuals = vector_sub(y, y_hat)
    rss = dot(residuals, residuals)
    
    n = len(X)
    p = len(X[0])
    sigma2_hat = rss / (n - p)
    
    return beta_hat, sigma2_hat
\end{lstlisting}

def hat_matrix(X):
    X_T = transpose(X)
    XtX = matmul(X_T, X)
    inv_XtX = inverse(XtX)
    H = matmul(matmul(X, inv_XtX), X_T)

\subsection{Đo lường chỉ số đánh giá: \texttt{model\_metrics}}
\begin{lstlisting}[language=Python]
def model_metrics(y, y_hat, p):
    n = len(y)
    residuals = vector_sub(y, y_hat)
    rss = dot(residuals, residuals)
    
    y_bar = sum(y) / n
    tss = sum((yi - y_bar) ** 2 for yi in y)
    
    r2 = 1.0 - (rss / tss)
    r2_adj = 1.0 - ((n - 1) / (n - p - 1)) * (1.0 - r2)
    
    f_stat = ((tss - rss) / p) / (rss / (n - p - 1))
    
    return rss, tss, r2, r2_adj, f_stat
\end{lstlisting}

\subsection{Tính toán chỉ số phóng đại phương sai: \texttt{vif}}
\begin{lstlisting}[language=Python]
def vif(X):
    n = len(X)
    p = len(X[0])
    if p == 1:
        return {"x1": 1.0}
    
    vif_dict = {}
    for j in range(p):
        y_j = [row[j] for row in X]
        X_others = [[row[i] for i in range(p) if i != j] for row in X]
        
        beta_sub, _ = ols_fit(X_others, y_j)
        y_hat_sub = matvec(X_others, beta_sub)
        
        res_sub = vector_sub(y_j, y_hat_sub)
        rss_sub = dot(res_sub, res_sub)
        
        yj_bar = sum(y_j) / n
        tss_sub = sum((val - yj_bar) ** 2 for val in y_j)
        
        r2_sub = 1.0 - (rss_sub / tss_sub)
        vif_dict[f"x{j}"] = 1.0 / (1.0 - r2_sub)
        
    return vif_dict
\end{lstlisting}

\subsection{Hiện thực mô hình điều chỉnh Ridge: \texttt{ridge\_fit}}
\begin{lstlisting}[language=Python]
def ridge_fit(X, y, lam):
    p = len(X[0])
    I = identity_matrix(p)
    I[0][0] = 0.0
    
    lam_I = [[lam * I[i][j] for j in range(p)] for i in range(p)]
    
    X_T = transpose(X)
    XtX = matmul(X_T, X)
    
    XtX_lam_I = [[XtX[i][j] + lam_I[i][j] for j in range(p)] for i in range(p)]
    inv_matrix = inverse(XtX_lam_I)
    Xty = matvec(X_T, y)
    
    beta_ridge = matvec(inv_matrix, Xty)
    return beta_ridge
\end{lstlisting}

def kfold_cv(X, y, k, lam):
    n = len(y)
    indices = list(range(n))
    random.shuffle(indices)
    
    fold_sizes = [n // k + (1 if i < n % k else 0) for i in range(k)]
    folds = []
    current = 0
    for size in fold_sizes:
        folds.append(indices[current:current + size])
        current += size
        
    mse_scores = []
    for i in range(k):
        test_indices = folds[i]
        train_indices = [idx for idx in indices if idx not in test_indices]
        
        X_train = [X[idx] for idx in train_indices]
        y_train = [y[idx] for idx in train_indices]
        X_test = [X[idx] for idx in test_indices]
        y_test = [y[idx] for idx in test_indices]
        
        beta_train = ridge_fit(X_train, y_train, lam)
        y_pred = matvec(X_test, beta_train)
        
        res = vector_sub(y_test, y_pred)
        mse = dot(res, res) / len(y_test)
        mse_scores.append(mse)
        
    return sum(mse_scores) / k
\end{lstlisting}

Dựa trên phân tích ở trên, nhóm áp dụng chiến lược xử lý ba bước như sau:

\textbf{Bước 1 --- Tự động loại bỏ các cột có tỉ lệ missing vượt ngưỡng 70\%.}
Pipeline tính tỉ lệ missing của từng cột trong tập train và loại bỏ bất kỳ cột nào vượt ngưỡng 70\%. Biến \texttt{NMHC(GT)} (90.23\%) bị loại theo quy tắc này mà không cần hardcode tên cột, đảm bảo tính tổng quát.

# Buoc 1: Tu dong loai cot co ti le missing > 70% (hoc tren train)
missing_pct = X_tmp.isnull().mean()
high_missing_cols = missing_pct[missing_pct > 0.7].index.tolist()
X_tmp = X_tmp.drop(columns=high_missing_cols)
\end{lstlisting}

\textbf{Bước 2 --- Loại bỏ các hàng thiếu giá trị biến mục tiêu.}
Các hàng thiếu \texttt{CO(GT)} bị loại trước khi chia tập train/test. Đây là bước bắt buộc vì không thể impute biến mục tiêu --- điều đó tương đương với việc mô hình tự dự báo chính nó. Sau bước này còn lại 7765 quan sát.

\begin{lstlisting}[language=Python]

\end{lstlisting}

\textbf{Bước 3 --- KNN Imputation cho các đặc trưng còn thiếu.}
Thay vì xóa toàn bộ hàng còn lại có giá trị NaN (listwise deletion), nhóm sử dụng phương pháp KNN Imputation ($k=5$) để phục hồi các giá trị thiếu trong nhóm cảm biến PT08.Sx và các biến khí hậu (tỉ lệ missing 3.91\%). Với mỗi hàng có NaN, pipeline tìm $k=5$ hàng gần nhất trong tập train dựa trên khoảng cách Euclidean tính trên các đặc trưng không bị thiếu, sau đó điền bằng giá trị trung bình của $k$ hàng đó. Trường hợp đặc biệt, nếu hàng có ít hơn $\texttt{MIN\_COMMON\_FEATURES} = 3$ đặc trưng hợp lệ, hệ thống fallback về median của cột. Toàn bộ tham số KNN (mean, std chuẩn hóa, dữ liệu train làm tham chiếu) được học \textbf{chỉ từ tập train}, đảm bảo không xảy ra data leakage sang tập test.

\begin{lstlisting}[language=Python]
# Buoc 3: KNN Imputation (k=5), hoc tren train, ap dung len ca test
# Chuan hoa truoc de tinh khoang cach cong bang giua cac features
X_scaled = (X - knn_means) / knn_stds
for idx, row_scaled in X_scaled.loc[missing_rows.index].iterrows():
    valid_features = row_scaled.dropna().index
    if len(valid_features) < MIN_COMMON_FEATURES:
        X_out.loc[idx, col] = medians_[col]  # fallback median
        continue
    sq_dist = ((X_train_scaled[valid_features] - row_scaled[valid_features]) ** 2).sum(axis=1)
    k_nearest = sq_dist.nsmallest(5).index
    X_out.loc[idx, col] = X_train_raw.loc[k_nearest, col].mean()
\end{lstlisting}

\textbf{Kết quả sau xử lý:} Sau ba bước trên, bộ dữ liệu còn lại $7765$ quan sát (tương đương $83.0\%$ dữ liệu gốc), được chia thành $6212$ mẫu train và $1553$ mẫu test theo tỉ lệ 80/20 --- đủ lớn để đảm bảo độ tin cậy thống kê cho các mô hình hồi quy.

\subsubsection{Chiến lược xử lý outliers}

\begin{itemize}
    \item Nhóm áp dụng \textbf{Winsorization} tại ngưỡng percentile 1\% và 99\% cho từng đặc trưng. Các giá trị nằm ngoài khoảng $[q_{0.01},\, q_{0.99}]$ được cắt cụt về đúng ngưỡng đó thay vì bị xóa hoàn toàn:
    \begin{enumerate}
        \item Winsorization giữ lại toàn bộ số lượng quan sát, tránh mất thông tin do xóa hàng.
        \item Bằng cách cắt cụt thay vì xóa, các sự kiện ô nhiễm đột ngột vẫn được phản ánh trong dữ liệu nhưng không kéo quá lớn hệ số hồi quy.
        \item Ngưỡng Winsorize ($q_{0.01}$, $q_{0.99}$) được học \textbf{chỉ từ tập train} rồi áp dụng lên tập test, đảm bảo không có data leakage.
    \end{enumerate}
\end{itemize}

\begin{lstlisting}[language=Python]
# Winsorization tai nguong 1% va 99% (hoc tren train, ap dung len test)
for col in numeric_cols:
    q01 = X_tmp[col].quantile(0.01)
    q99 = X_tmp[col].quantile(0.99)
    winsorize_bounds[col] = (q01, q99)
    X_tmp[col] = X_tmp[col].clip(lower=q01, upper=q99)
\end{lstlisting}
 
\subsection{Chuẩn hóa dữ liệu (Standardization)}

\paragraph{Ý nghĩa của bước chuẩn hóa dữ liệu}

Việc chuẩn hóa dữ liệu trước khi huấn luyện là cần thiết vì các đặc trưng trong bộ dữ liệu tồn tại trên nhiều thang đo khác nhau. Các tín hiệu cảm biến PT08.Sx có giá trị ở mức hàng nghìn, trong khi nhiệt độ và độ ẩm chỉ dao động trong khoảng vài chục đơn vị. Nếu giữ nguyên thang đo ban đầu, các biến có magnitude lớn sẽ chi phối quá trình tối ưu hóa và làm sai lệch ý nghĩa của hệ số hồi quy.

Ngoài ra, Ridge Regression và Lasso Regression đều sử dụng regularization phụ thuộc trực tiếp vào độ lớn của hệ số. Do đó, việc chuẩn hóa giúp quá trình regularization tác động công bằng hơn giữa các biến, đồng thời cải thiện độ đồng thời giúp quá trình tối ưu ổn định hơn khi huấn luyện mô hình.

\paragraph{Phương pháp chuẩn hóa Z-score}

Nhóm áp dụng chuẩn hóa Z-score (Standard Scaler) cho tất cả các đặc trưng số. Với mỗi đặc trưng $j$, giá trị chuẩn hóa $z_{ij}$ được tính theo công thức:

\[
z_{ij} = \frac{x_{ij} - \hat{\mu}_j}{\hat{\sigma}_j}
\]

trong đó $\hat{\mu}_j$ và $\hat{\sigma}_j$ là giá trị trung bình và độ lệch chuẩn (với $\text{ddof}=1$) của đặc trưng $j$ \textbf{chỉ tính trên tập train}. Sau đó cùng một cặp $(\hat{\mu}_j, \hat{\sigma}_j)$ được áp dụng lên tập test mà không tính lại --- đây là điều kiện bắt buộc để tránh data leakage.

\begin{lstlisting}[language=Python]
# Chuan hoa Z-score: hoc mean/std tren train, ap dung ca train va test
n = len(X_train)
for j in range(len(X_train[0])):
    col = [X_train[i][j] for i in range(n)]
    mu = sum(col) / n
    std = (sum((v - mu) ** 2 for v in col) / (n - 1)) ** 0.5
    scaler_params[j] = (mu, std if std > EPSILON else 1.0)

# Ap dung len ca train va test
X_train_scaled = [[(X_train[i][j] - scaler_params[j][0]) / scaler_params[j][1]
                   for j in range(p)] for i in range(n_train)]
X_test_scaled  = [[(X_test[i][j]  - scaler_params[j][0]) / scaler_params[j][1]
                   for j in range(p)] for i in range(n_test)]
\end{lstlisting}

\textbf{Kết quả:} Sau chuẩn hóa, tất cả 11 đặc trưng đều có $\hat{\mu} = 0$ và $\hat{\sigma} = 1$ trên tập train. Đặc trưng \texttt{EPSILON} ($= 10^{-12}$) được dùng làm guard để tránh chia cho 0 khi cột có phương sai bằng không.

\subsection{Mở rộng đặc trưng đa thức (Polynomial Feature Expansion)}

\subsubsection{Động lực và ý nghĩa}

Mối quan hệ giữa tín hiệu cảm biến và nồng độ CO không nhất thiết là tuyến tính thuần túy. Chẳng hạn, tương tác giữa nhiệt độ và độ ẩm có thể ảnh hưởng phi tuyến đến phản ứng cảm biến. Mở rộng đặc trưng bậc hai cho phép mô hình tuyến tính nắm bắt các mối quan hệ phi tuyến và tương tác bậc hai mà không cần thay đổi thuật toán hồi quy.

\subsubsection{Công thức mở rộng bậc 2}

Với $n$ đặc trưng gốc, mở rộng bậc 2 sinh ra tất cả các đơn thức bậc $\leq 2$:

\[
\Phi(\mathbf{x}) = \bigl[\,x_1,\; x_2,\; \ldots,\; x_n,\; x_1^2,\; x_1 x_2,\; x_1 x_3,\; \ldots,\; x_n^2\,\bigr]
\]

Số đặc trưng sau mở rộng (không tính bias) là $\binom{n+2}{2} - 1 = \dfrac{(n+1)(n+2)}{2} - 1$.

Với $n = 11$ đặc trưng gốc (sau khi loại bỏ \texttt{NMHC(GT)}, \texttt{Date}, \texttt{Time} và biến mục tiêu):
\[
\frac{12 \times 13}{2} - 1 = 78 - 1 = 77 \text{ đặc trưng}
\]

\subsubsection{Cài đặt}

\begin{lstlisting}[language=Python]
# Mo rong dac trung bac 2: tu 11 -> 77 features
def poly_features(X, degree=2):
    n, p = len(X), len(X[0])
    result = []
    for row in X:
        expanded = list(row)                  # n features bac 1
        for i in range(p):
            expanded.append(row[i] ** 2)      # x_i^2
            for j in range(i + 1, p):
                expanded.append(row[i] * row[j])  # x_i * x_j
        result.append(expanded)
    return result

X_train_poly = poly_features(X_train_scaled)  # 6212 x 77
X_test_poly  = poly_features(X_test_scaled)   # 1553 x 77
\end{lstlisting}

\textbf{Kết quả:} Tập dữ liệu cuối cùng có kích thước $6212 \times 77$ (train) và $1553 \times 77$ (test). Số đặc trưng tăng từ 11 lên 77, tạo không gian biểu diễn phong phú hơn cho các mô hình hồi quy.

\end{itemize}
